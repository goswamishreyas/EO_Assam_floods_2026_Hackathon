// Function to date as a property of the image
var add_date = function(image)
{
  var system_date = ee.Date(image.get('system:time_start')).format().slice(0,10);
  image = image.set({'Date':system_date})
  return image
}

var new_coll = image_coll.map(add_date)
print(new_coll)
// Get a list of unique dates in the image collection
var date_list = new_coll.aggregate_array('Date').distinct()
print(date_list)
// print(ee.Date(date_list.get(4)))

// Function to create mosaic from the images for each unique date
var mosaic_unique_date = function(date)    
{
  date = ee.Date(date)
  var mosaic = new_coll.select(['B2','B8','B4','B3','B11']).filterDate(date,date.advance(1,'day')).mosaic().divide(10000).clip(aoi)
  var ndvi = mosaic.normalizedDifference(['B8','B4']).rename('NDVI');
  var ndmi = mosaic.normalizedDifference(['B8','B11']).rename('NDMI');
  var ndwi = mosaic.normalizedDifference(['B3', 'B8']).rename('NDWI');
  var mndwi = mosaic.normalizedDifference(['B3','B11']).rename('MNDWI');
  mosaic = mosaic.addBands([ndvi,ndmi,ndwi,mndwi]);
  mosaic = mosaic.set({
    'system:index': date.format().slice(0,10),
    'Date':date
  })
  return mosaic.select(['B11','B8','B4','B3','B2','NDVI','NDMI','NDWI','MNDWI']);
}

// Creating a collection of mosaic from each unique date of the image collection
var mosaic_coll = date_list.map(mosaic_unique_date)
